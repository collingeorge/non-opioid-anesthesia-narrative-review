\begin{table}[H]

\caption{Premedication Strategies: Context-Dependent Selection for NOA Protocols}
\label{tab:s1-premed}
\small
\begin{tabular}{@{}p{3.5cm}p{5.5cm}p{5.5cm}@{}}
\toprule
\textbf{Feature} & \textbf{Midazolam}\newline\textbf{(Standard Anxiolysis)} & \textbf{Dexmedetomidine}\newline\textbf{(NOA-Integrated Premedication)} \\
\midrule
\textbf{Clinical Context} & Routine preoperative anxiolysis in stable patients & Comprehensive NOA protocols requiring multimodal analgesia \\
\addlinespace
\textbf{Primary Goal} & Rapid anxiolysis \& amnesia & Anxiolysis with opioid-sparing analgesia \\
\addlinespace
\textbf{Onset Time} & Rapid (1--2 min IV) & Slower (10--15 min) \\
\addlinespace
\textbf{Monitoring Requirements} & Minimal; safe for immediate transport & Continuous HR/BP monitoring required \\
\addlinespace
\textbf{Respiratory Effect} & Mild depression (dose-dependent) & Minimal to none \\
\addlinespace
\textbf{Hemodynamic Effects} & Minimal in healthy patients & Bradycardia (5--15\%), hypotension (10--20\%) \\
\addlinespace
\textbf{Postoperative Delirium} & May increase risk (elderly) & May reduce risk \\
\addlinespace
\textbf{Opioid-Sparing Effect} & None & Significant (20--40\% reduction) \\
\addlinespace
\textbf{Ease of Use} & Comfortable for all clinicians; straightforward dosing & Requires experience; "finicky" dosing \\
\addlinespace
\textbf{Ideal Use Case} & Routine anxiolysis, time-sensitive cases, immediate transport needed & Comprehensive NOA protocols, high OUD risk, planned monitored recovery \\
\bottomrule
\end{tabular}
\medskip

\small\textit{Note:} Midazolam remains the standard preoperative anxiolytic for routine use due to rapid onset, ease of administration, and safety without continuous monitoring. Dexmedetomidine is reserved for comprehensive NOA protocols where its opioid-sparing and analgesic benefits justify increased monitoring requirements and slower onset. Selection should be guided by clinical context, patient risk factors, surgical complexity, and institutional resources \citep{Mena2022}.
\end{table}
